# ✅ 审核队列

> [!info] 审核、主题与批次是三件事
> **批次**只记录共同作者或一次处理任务，不决定正式库目录；**建议主题目录**依据每篇论文本身的研究对象和方法逐篇确定；只有完成笔记内“快速审核”清单后，才允许快速入库。证据未完成时明确显示“待证据审核”，不会再写成“入库后继续核验”。

## 待处理文献

```dataviewjs
const root = dv.container;
const message = root.createEl("div");
message.style.cssText = "margin:8px 0;color:var(--text-muted)";
const controls = root.createDiv();
controls.style.cssText = "display:flex;gap:8px;margin:10px 0;flex-wrap:wrap";
const importEligible = controls.createEl("button", {text:"📥 批量入库已完成快速审核"});
const refreshButton = controls.createEl("button", {text:"刷新队列"});
const list = root.createDiv();
const prop = (obj, cn, legacy) => obj?.[cn] ?? obj?.[legacy];
const setProp = (obj, cn, legacy, value) => {
  if (Object.prototype.hasOwnProperty.call(obj, cn) || !Object.prototype.hasOwnProperty.call(obj, legacy)) obj[cn] = value;
  else obj[legacy] = value;
};

function draftPages() {
  const pages = dv.pages('"01_收件箱/AI草稿"')
    .where(p => ["文献", "literature"].includes(prop(p, "类型", "type")) && !p.file.path.includes("/_原始输出/"))
    .sort(p => p.file.mtime, "desc").array();
  const seen = new Set();
  return pages.filter(p => {
    const key = prop(p, "Zotero条目键", "zotero_item_key"), doi = prop(p, "DOI", "doi");
    const identity = key ? `zotero:${key}` : doi ? `doi:${String(doi).toLowerCase()}` : `file:${p.file.path}`;
    if (seen.has(identity)) return false;
    seen.add(identity); return true;
  });
}
function safeFolder(value) {
  const s = String(value || "未分类").trim();
  return s && !s.includes("..") && !s.includes("/") && !s.includes("\\") ? s : null;
}
function quickState(content) {
  const block = content.match(/<!-- REVIEW:QUICK:BEGIN -->([\s\S]*?)<!-- REVIEW:QUICK:END -->/);
  if (!block) return {total:0, done:0, ready:false};
  const tasks = [...block[1].matchAll(/^(?:>\s*)?- \[([ xX])\]/gm)];
  const done = tasks.filter(task => task[1].toLowerCase() === "x").length;
  return {total:tasks.length, done, ready:tasks.length > 0 && done === tasks.length};
}
function firstImage(content, sourcePath) {
  const link = content.match(/!\[\[([^\]|]+)(?:\|[^\]]+)?\]\]/)?.[1];
  if (!link) return null;
  return app.metadataCache.getFirstLinkpathDest(link, sourcePath);
}
async function pageInfo(page) {
  const file = app.vault.getAbstractFileByPath(page.file.path);
  const content = file ? await app.vault.cachedRead(file) : "";
  return {file, content, quick:quickState(content), image:firstImage(content, page.file.path)};
}
async function quickImport(page, ask = true) {
  const info = await pageInfo(page);
  const file = info.file;
  if (!file) return {ok:false, reason:"源文件不存在"};
  const processState = prop(page, "处理状态", "processing_status");
  if (processState && !["完整", "complete"].includes(processState)) return {ok:false, reason:"处理尚未完整完成"};
  if (!info.quick.ready) return {ok:false, reason:`快速审核清单尚未完成（${info.quick.done}/${info.quick.total}）`};
  if (!info.image) return {ok:false, reason:"没有可解析的原文图片"};
  const folder = safeFolder(prop(page, "入库目录", "library_folder"));
  if (!folder) return {ok:false, reason:"建议主题目录无效"};
  const targetFolder = `20_文献/${folder}`;
  let targetPath = `${targetFolder}/${file.name}`;
  const existing = app.vault.getAbstractFileByPath(targetPath);
  let replaceSamePaper = false;
  let conflictNote = "";
  if (existing) {
    const oldFm = app.metadataCache.getFileCache(existing)?.frontmatter || {};
    const cleanDoi = value => String(value || "").toLowerCase().replace(/^https?:\/\/(dx\.)?doi\.org\//, "").trim();
    const pageKey = prop(page, "Zotero条目键", "zotero_item_key"), oldKey = prop(oldFm, "Zotero条目键", "zotero_item_key");
    const pageDoi = prop(page, "DOI", "doi"), oldDoi = prop(oldFm, "DOI", "doi");
    const sameKey = pageKey && oldKey && String(pageKey) === String(oldKey);
    const sameDoi = cleanDoi(pageDoi) && cleanDoi(pageDoi) === cleanDoi(oldDoi);
    replaceSamePaper = Boolean(sameKey || sameDoi);
    if (replaceSamePaper) conflictNote = "\n\n正式库中已有同一文献：旧版将备份后由新版替换。";
    else {
      const base = file.name.replace(/\.md$/i, ""); let n = 1;
      targetPath = `${targetFolder}/${base}-重解析版.md`;
      while (app.vault.getAbstractFileByPath(targetPath)) targetPath = `${targetFolder}/${base}-重解析版-${++n}.md`;
      conflictNote = `\n\n存在无法确认身份的同名文件，新版将另存为 ${targetPath.split("/").pop()}。`;
    }
  }
  if (ask && !window.confirm(`已完成快速审核。确认移动到逐篇主题目录：\n${targetPath}？${conflictNote}`)) return {ok:false, reason:"已取消"};
  try {
    const today = window.moment ? window.moment().format("YYYY-MM-DD") : new Date().toISOString().slice(0,10);
    await app.fileManager.processFrontMatter(file, fm => {
      setProp(fm, "状态", "status", "已快速审核"); setProp(fm, "审核等级", "review_level", "快速审核");
      setProp(fm, "快速审核日期", "quick_reviewed_date", today); setProp(fm, "更新时间", "updated", today);
      setProp(fm, "证据状态", "evidence_status", "待审核");
    });
    if (!app.vault.getAbstractFileByPath(targetFolder)) await app.vault.createFolder(targetFolder);
    if (existing && replaceSamePaper) {
      const parent = "99_系统/迁移/同名冲突备份";
      const dateFolder = `${parent}/${today}`;
      if (!app.vault.getAbstractFileByPath(parent)) await app.vault.createFolder(parent);
      if (!app.vault.getAbstractFileByPath(dateFolder)) await app.vault.createFolder(dateFolder);
      let backupPath = `${dateFolder}/${existing.name}`;
      if (app.vault.getAbstractFileByPath(backupPath)) backupPath = `${dateFolder}/${existing.basename}-${Date.now()}.md`;
      await app.vault.copy(existing, backupPath); await app.vault.delete(existing);
    }
    await app.fileManager.renameFile(file, targetPath);
    return {ok:true, targetPath};
  } catch (error) { return {ok:false, reason:error.message || String(error)}; }
}
async function render() {
  const pages = draftPages();
  message.textContent = `待审核：${pages.length} 篇`;
  list.empty();
  if (!pages.length) { list.createEl("p", {text:"当前没有待处理文献。"}); return; }
  const table = list.createEl("table");
  const head = table.createEl("thead").createEl("tr");
  ["原文图","文献","处理批次","建议主题目录","审核状态","操作"].forEach(text => head.createEl("th", {text}));
  const body = table.createEl("tbody");
  for (const page of pages) {
    const info = await pageInfo(page);
    const row = body.createEl("tr");
    const preview = row.createEl("td");
    if (info.image) {
      const img = preview.createEl("img");
      img.src = app.vault.getResourcePath(info.image);
      img.style.cssText = "width:96px;height:64px;object-fit:cover;border-radius:3px";
    } else preview.createEl("span", {text:"缺图"});
    const titleCell = row.createEl("td");
    const link = titleCell.createEl("a", {text:prop(page, "标题", "title") || page.file.name, cls:"internal-link"});
    link.setAttr("data-href", page.file.path); link.setAttr("href", page.file.path);
    titleCell.createEl("div", {text:String(prop(page, "Zotero条目键", "zotero_item_key") || "未匹配")}).style.cssText = "font-size:11px;color:var(--text-muted)";
    row.createEl("td", {text:prop(page, "处理批次", "batch_name") || page.file.folder.split("/").pop() || "未记录"});
    row.createEl("td", {text:String(prop(page, "入库目录", "library_folder") || "未分类")});
    const verified = prop(page, "证据状态", "evidence_status") === "已核验" || (page.figures_verified === true && page.tables_verified === true);
    const stateText = !info.image ? "缺少原文图片" : verified ? "图表证据已核验" : `待证据审核 · 快速清单 ${info.quick.done}/${info.quick.total}`;
    row.createEl("td", {text:stateText});
    const action = row.createEl("td");
    const button = action.createEl("button", {text:info.quick.ready ? "快速审核入库" : "先完成快速审核"});
    const processState = prop(page, "处理状态", "processing_status");
    button.disabled = !info.quick.ready || !info.image || (processState && !["完整", "complete"].includes(processState));
    button.addEventListener("click", async () => {
      button.disabled = true;
      const result = await quickImport(page, true);
      new Notice(result.ok ? `已入库：${result.targetPath}` : `未入库：${result.reason}`);
      await render();
    });
  }
}
importEligible.addEventListener("click", async () => {
  const pages = draftPages(); const eligible = [];
  for (const page of pages) { const info = await pageInfo(page); const processState = prop(page, "处理状态", "processing_status"); if (info.quick.ready && info.image && (!processState || ["完整", "complete"].includes(processState))) eligible.push(page); }
  if (!eligible.length) { new Notice("当前没有完成快速审核且具备原文图片的笔记"); return; }
  if (!window.confirm(`确认将已完成快速审核的 ${eligible.length} 篇文献按各自建议主题目录入库？`)) return;
  let ok = 0; const failed = [];
  for (const page of eligible) { const result = await quickImport(page, false); if (result.ok) ok++; else failed.push(`${page.file.name}：${result.reason}`); }
  new Notice(`批量入库：成功 ${ok} 篇，跳过 ${failed.length} 篇`); await render();
});
refreshButton.addEventListener("click", render);
await render();
```

## 已快速审核但证据待审核

```dataview
TABLE WITHOUT ID file.link AS "文献", default(入库目录, library_folder) AS "主题目录", default(Zotero条目键, zotero_item_key) AS "Zotero", default(审核等级, review_level) AS "审核等级", default(更新时间, updated) AS "更新"
FROM "20_文献"
WHERE (类型 = "文献" OR type = "literature") AND (证据状态 = "待审核" OR evidence_pending = true)
SORT file.mtime DESC
```

## 已完成证据审核

```dataview
TABLE WITHOUT ID file.link AS "文献", default(入库目录, library_folder) AS "主题目录", default(审核等级, review_level) AS "审核等级", default(证据审核日期, evidence_reviewed_date) AS "证据审核日期"
FROM "20_文献"
WHERE (类型 = "文献" OR type = "literature") AND (审核等级 = "证据审核" OR 审核等级 = "深度审核" OR review_level = "evidence-reviewed" OR review_level = "deep-reviewed")
SORT file.mtime DESC
```
